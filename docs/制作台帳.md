# 制作台帳（ledger.json）の書式

状態: 叩き台（2026-11-18）。M1 で JSON Schema にする。
チャンネル側が書き出し、movie が読む。**この文書が書式の正本。**

```jsonc
{
  "ledger_version": "0.1",
  "episode": {
    "id": "EP001", "channel": "minecraft_physics",
    "title": "...", "target_sec": 600,
    "media_root": "D:/mcphys/media/EP001",       // 素材のある場所（git外）。下のパスはここからの相対パス
    "approvals": { "theme": {...}, "design": {...}, "publish": null }
  },
  "cast": [                                        // 掛け合いの話者
    { "id": "host",  "role": "解説役", "tts": { "engine": "voicevox", "speaker": 0 }, "portrait": "cast/host/" },
    { "id": "guest", "role": "聞き手", "tts": { "engine": "voicevox", "speaker": 0 }, "portrait": "cast/guest/" }
  ],
  "sections": [                                    // フック / 予想 / 実験 / 結果 / 発展
    {
      "id": "hook", "kind": "hook",
      "lines": [
        { "id": "l001", "speaker": "guest", "text": "金床と豚、同時に落としたらどっちが先？",
          "reading": null,                         // 読みの上書き（誤読対策）
          "emphasis": ["金床", "豚"] }             // 字幕で強調する語
      ],
      "visuals": [
        { "id": "v001", "kind": "footage", "file": "shots/E001/landing_close.mp4",
          "sync": { "start_line": "l001", "offset_sec": 0.0 }, "speed": 1.0 },
        { "id": "v002", "kind": "chart", "data": "data/E001/measurements.csv",
          "chart": { "type": "line_vs_points", "x": "t_sec", "y": "y", "series": ["anvil", "pig", "item"], "theory": "data/E001/theory.csv" },
          "sync": { "start_line": "l003" } },
        { "id": "v003", "kind": "equation", "tex": "v_{n+1} = (v_n - g)(1-d)", "sync": { "start_line": "l005" } }
      ],
      "highlight": { "score": 0.9, "reason": "結果が一番意外な場面" }   // ショートの素材候補
    }
  ],
  "facts": [                                       // 台本中の数字や主張と、その根拠
    { "id": "f01", "text": "生物の重力加速度は 32 m/s²", "confidence": "実測", "source": "experiments/E001/results/..." }
  ],
  "music": { "bgm": [...], "sfx": [...] },
  "outputs": {
    "long":   { "aspect": "16:9", "file": null },
    "shorts": [ { "id": "S1", "aspect": "9:16", "source_sections": ["hook", "result"], "file": null } ]
  },
  "qc": { "findings": [] }
}
```

## 決まり

- **`facts` の `confidence` が【実測】【一次】以外の主張は、台本で断言しない。** 検査で、台詞に含まれる数字が `facts` にあるかを照合する。
- `visuals[].sync` は台詞のIDで指定する。秒数は音声を作ったあとに movie が計算する（台本を直しても画面がずれない）。
- 素材のパスは `media_root` からの相対パス。台帳自体はチャンネル側のリポジトリにコミットしてよい。
